Calculation:
  gross_need =
    max(0, boat_target - Shopify boat available)
    + max(0, cornwall_target - Shopify cornwall available)

  already_in_forge =
    assembled
    + awaiting_dispatch
    + in_transit_cornwall
    + intact_rework

  need_to_make = max(0, gross_need - already_in_forge)
